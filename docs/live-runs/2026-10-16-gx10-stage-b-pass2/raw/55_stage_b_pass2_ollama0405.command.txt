import json, sys, pathlib

# Read-only: finds the harness's in-process gateway port (lsof on the harness python process, matched by the
# venv python path) and polls /admin/status every 5 s. Records health flags, last_refresh_error, registry_version,
# request counters. The admin key is read from the key file for the Authorization header and never written anywhere.
import json, os, re, subprocess, sys, time, urllib.request
home, out, pypath = sys.argv[1:4]
key = open(os.path.join(home, "keys", "admin_key")).read().split()[0]
me = str(os.getpid())
def harness_pids():
    o = subprocess.run(["pgrep", "-f", pypath], capture_output=True, text=True).stdout.split()
    return [p for p in o if p != me]
def ports(pids):
    if not pids: return [], ""
    o = subprocess.run(["lsof", "-nP", "-a", "-iTCP", "-sTCP:LISTEN", "-p", ",".join(pids)], capture_output=True, text=True).stdout
    return sorted({int(m) for m in re.findall(r":(\d+) \(LISTEN\)", o)}), o
seen_any = False; deadline = time.time() + 3 * 3600; last_diag = 0
with open(out, "a") as f:
    while time.time() < deadline:
        pids = harness_pids()
        if not pids and seen_any: break
        ps, raw = ports(pids)
        if time.time() - last_diag >= 60:
            print(time.strftime("%H:%M:%S"), f"harness pids {pids} listen ports {ps}" + ("" if ps or not pids else f" lsof: {raw.strip()[:300]!r}"), flush=True); last_diag = time.time()
        for p in ps:
            try:
                req = urllib.request.Request(f"http://127.0.0.1:{p}/admin/status", headers={"Authorization": f"Bearer {key}"})
                s = json.load(urllib.request.urlopen(req, timeout=3)); seen_any = True
                rec = {"t": time.strftime("%Y-%m-%dT%H:%M:%S%z"), "port": p, "ready": s.get("ready"), "registry_version": s.get("registry_version"),
                       "last_refresh_error": s.get("last_refresh_error"), "requests_total": s.get("requests_total"), "requests_failed": s.get("requests_failed"),
                       "deployments": {d["id"]: {"healthy": d["state"]["healthy"], "loaded": d["state"]["loaded"]} for d in s.get("deployments", [])}}
                f.write(json.dumps(rec) + "\n"); f.flush()
                if rec["deployments"] and not all(v["healthy"] for v in rec["deployments"].values()):
                    print(rec["t"], "UNHEALTHY", rec["last_refresh_error"], flush=True)
            except Exception as e:
                if time.time() - last_diag < 1: print("  poll error on port", p, repr(e)[:120], flush=True)
        time.sleep(5)
